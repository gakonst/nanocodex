import createCore from './core.js';
import wasm from './core.wasm';

// One command per fresh isolate. The caller supplies only the selected input,
// never a filesystem/network binding or credentials.
export default {
  async fetch(request) {
    const form = await request.formData();
    const args = JSON.parse(form.get('args'));
    const input = form.get('input');
    const inputPath = form.get('inputPath');
    const outputPath = form.get('outputPath');
    if (!Array.isArray(args) || args.some(arg => typeof arg !== 'string')) {
      return new Response('Invalid media arguments', {status: 400});
    }
    let stdout = '', stderr = '';
    const append = (stream, line) => {
      const value = `${line}\n`;
      if (stream === 'stdout') stdout += value; else stderr += value;
    };
    try {
      const core = await createCore({
        noInitialRun: true,
        print: line => append('stdout', line),
        printErr: line => append('stderr', line),
        instantiateWasm(imports, receive) {
          const instance = new WebAssembly.Instance(wasm, imports);
          receive(instance, wasm);
          return instance.exports;
        },
      });
      if (input !== null) {
        if (!(input instanceof File) || !/^\/input\.[a-z0-9]+$/.test(inputPath)) throw new Error('Invalid media input');
        core.FS.writeFile(inputPath, new Uint8Array(await input.arrayBuffer()));
      }
      if (outputPath !== null) {
        if (!/^\/output\.[a-z0-9]+$/.test(outputPath)) throw new Error('Invalid media output');
      }
      // FS.init closes stdout only after callMain returns; EXIT_RUNTIME flushes
      // partial lines and preserves FFmpeg's normal status codes.
      let exitCode = core.callMain(args);
      if (!Number.isInteger(exitCode)) exitCode = 1;
      const result = new FormData();
      result.set('result', JSON.stringify({stdout, stderr, exitCode}));
      if (exitCode === 0 && outputPath !== null) {
        const data = core.FS.readFile(outputPath);
        result.set('output', new Blob([data]), outputPath.slice(1));
      }
      return new Response(result);
    } catch (error) {
      return Response.json({error: error instanceof Error ? error.message : 'Media execution failed'}, {status: 422});
    }
  },
};
